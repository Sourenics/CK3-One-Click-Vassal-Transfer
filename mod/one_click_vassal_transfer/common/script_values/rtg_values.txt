# Nº de condados del subreino de este personaje que están de iure
# dentro de scope:rtg_title. Se usa como plan B para elegir receptor.
rtg_counties_in_title_value = {
	value = 0
	every_sub_realm_county = {
		limit = { target_is_de_jure_liege_or_above = scope:rtg_title }
		add = 1
	}
}

# Nº de condados que "Otorgar condados" entregaría (= nuevos vasallos).
rtg_counties_to_grant_value = {
	value = 0
	every_held_title = {
		limit = { rtg_can_grant_county_trigger = yes }
		add = 1
	}
}

# Desempate por personalidad: 0–99, base 50.
rtg_loyalty_traits_value = {
	value = 50
	if = { limit = { has_trait = loyal } add = 30 }
	if = { limit = { has_trait = content } add = 10 }
	if = { limit = { has_trait = trusting } add = 5 }
	if = { limit = { has_trait = humble } add = 3 }
	if = { limit = { has_trait = just } add = 2 }
	if = { limit = { has_trait = disloyal } add = -30 }
	if = { limit = { has_trait = ambitious } add = -15 }
	if = { limit = { has_trait = arrogant } add = -5 }
	if = { limit = { has_trait = greedy } add = -3 }
	if = { limit = { has_trait = fickle } add = -2 }
	min = 0
	max = 99
}

# Orden del último recurso al repartir títulos (scope: candidato).
#   1) más condados dentro de scope:rtg_title
#   2) mejor opinión hacia root (tramos de 10)
#   3) rasgos de lealtad
#   4) diplomacia
rtg_recipient_score_value = {
	value = rtg_counties_in_title_value
	multiply = 100000
	add = {
		value = rtg_opinion_of_root_value
		multiply = 100
	}
	add = rtg_loyalty_traits_value
	add = {
		value = diplomacy
		divide = 100
	}
}
