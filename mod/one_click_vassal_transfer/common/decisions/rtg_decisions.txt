# 1. Transferir vasallos a su señor de iure
rtg_transfer_vassals_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
	desc = rtg_transfer_vassals_decision_desc
	selection_tooltip = rtg_transfer_vassals_decision_tooltip
	decision_group_type = rtg

	is_shown = {
		is_ai = no
		highest_held_title_tier >= tier_duchy
		# Oculta la decisión si no hay nada que hacer
		any_vassal = { rtg_vassal_would_transfer_trigger = yes }
	}
	is_valid = {
		any_vassal = { is_landed = yes }
	}
	effect = {
		custom_tooltip = rtg_transfer_vassals_decision_effect_tt
		show_as_tooltip = { rtg_preview_transfer_effect = yes }
		hidden_effect = { rtg_transfer_vassals_to_de_jure_liege_effect = { LIEGE = root } }
	}
	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
