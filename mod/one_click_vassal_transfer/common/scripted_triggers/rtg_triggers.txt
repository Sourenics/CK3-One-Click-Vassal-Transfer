# =====================================================================
# Reparto de Títulos - triggers
# =====================================================================

# Scope: título. Excluye los títulos que el juego gestiona por su cuenta:
# títulos de familia noble (gobiernos administrativo, celestial, sōryō y
# ritsuryō), de nómadas y otros títulos sin tierras, compañías de
# mercenarios y órdenes sagradas.
rtg_is_regular_landed_title_trigger = {
	is_landless_type_title = no
	is_noble_family_title = no
	is_nomad_title = no
	is_mercenary_company = no
	is_holy_order = no
}

# Scope: título que root tiene. ¿Se puede repartir?
# Nunca el título principal ni el que contiene de iure tu capital.
rtg_can_distribute_title_trigger = {
	rtg_is_regular_landed_title_trigger = yes
	is_titular = no
	NOT = { has_variable = rtg_excluded_from_distribution }
	NOT = { this = root.primary_title }
	NOT = {
		any_de_jure_county = { this = root.capital_province.county }
	}
}

# Scope: condado que root tiene. ¿Se puede otorgar a un nuevo vasallo?
rtg_can_grant_county_trigger = {
	tier = tier_county
	rtg_is_regular_landed_title_trigger = yes
	NOT = { has_variable = rtg_excluded_from_distribution }
	NOT = { this = root.capital_province.county }
}

# Scope: vasallo directo de root. ¿Puede recibir un título repartido?
rtg_valid_recipient_trigger = {
	is_landed = yes
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	NOT = { is_at_war_with = root }
	# Un teócrata solo recibe títulos si tú también lo eres
	OR = {
		NOT = { government_has_flag = government_is_theocracy }
		root = { government_has_flag = government_is_theocracy }
	}
}

# Scope: candidato a recibir scope:rtg_title. Para evitar bordergore,
# tanto su capital (donde reside) como su título principal deben estar de
# iure dentro del título. Un duque de Baviera que vive en Sajonia no
# cumple para el reino de Baviera.
rtg_anchored_in_title_trigger = {
	exists = capital_province
	capital_province.county = { target_is_de_jure_liege_or_above = scope:rtg_title }
	exists = primary_title
	primary_title = { target_is_de_jure_liege_or_above = scope:rtg_title }
}

# Scope: vasallo. ¿Se le puede cambiar de señor sin romper nada?
rtg_can_be_moved_trigger = {
	NOT = { is_at_war_with = root }
	NOT = { any_vassal_or_below = { is_at_war_with = root } }
	# El líder de una confederación de casas (nómadas) no se mueve
	NAND = {
		is_house_head = yes
		exists = house.house_confederation
		house.house_confederation.leading_house = house
	}
}

# Scope: personaje candidato a nuevo señor de scope:rtg_vassal,
# dentro del reino de scope:rtg_liege.
rtg_valid_new_liege_trigger = {
	NOT = { this = scope:rtg_vassal }
	NOT = { this = scope:rtg_liege }
	exists = liege
	liege = scope:rtg_liege
	highest_held_title_tier > scope:rtg_vassal.highest_held_title_tier
	NOR = {
		is_at_war_with = scope:rtg_liege
		is_at_war_with = scope:rtg_vassal
	}
}

# Scope: vasallo directo de root. ¿"Transferir vasallos" lo movería?
# Misma lógica que rtg_find_de_jure_liege_effect, en forma de trigger,
# para ocultar la decisión cuando no hay nada que transferir.
rtg_vassal_would_transfer_trigger = {
	is_landed = yes
	exists = primary_title.de_jure_liege
	rtg_can_be_moved_trigger = yes
	save_temporary_scope_as = rtg_vassal
	root = { save_temporary_scope_as = rtg_liege }
	OR = {
		primary_title.de_jure_liege.holder ?= { rtg_valid_new_liege_trigger = yes }
		primary_title.de_jure_liege.de_jure_liege.holder ?= { rtg_valid_new_liege_trigger = yes }
	}
}

# ---------------------------------------------------------------------
# ARREGLAR BORDERGORE
# La "región" de un vasallo son los títulos de duque o superior que
# posee (normales, no titulares). Requieren scope:rtg_bg_holder.
# ---------------------------------------------------------------------

# Scope: título. ¿Está de iure dentro de la región de scope:rtg_bg_holder?
rtg_bg_title_in_region_trigger = {
	save_temporary_scope_as = rtg_bg_checked
	scope:rtg_bg_holder = {
		any_held_title = {
			tier > tier_county
			rtg_is_regular_landed_title_trigger = yes
			is_titular = no
			save_temporary_scope_as = rtg_bg_region
			scope:rtg_bg_checked = { target_is_de_jure_liege_or_above = scope:rtg_bg_region }
		}
	}
}

# Scope: vasallo directo de root. ¿Tiene condados propios fuera de su
# región y, a la vez, al menos uno dentro (para no dejarlo sin tierras)?
rtg_bg_has_outside_counties_trigger = {
	highest_held_title_tier > tier_county
	rtg_can_be_moved_trigger = yes
	save_temporary_scope_as = rtg_bg_holder
	any_held_title = {
		tier = tier_county
		rtg_is_regular_landed_title_trigger = yes
		rtg_bg_title_in_region_trigger = yes
	}
	any_held_title = {
		tier = tier_county
		rtg_is_regular_landed_title_trigger = yes
		NOT = { rtg_bg_title_in_region_trigger = yes }
	}
}

# Scope: vasallo directo de root. ¿Tiene vasallos cuyo título principal
# está fuera de su región?
rtg_bg_has_outside_subvassals_trigger = {
	highest_held_title_tier > tier_county
	rtg_can_be_moved_trigger = yes
	save_temporary_scope_as = rtg_bg_holder
	any_vassal = {
		is_landed = yes
		rtg_can_be_moved_trigger = yes
		primary_title = { NOT = { rtg_bg_title_in_region_trigger = yes } }
	}
}

rtg_bg_vassal_has_bordergore_trigger = {
	OR = {
		rtg_bg_has_outside_counties_trigger = yes
		rtg_bg_has_outside_subvassals_trigger = yes
	}
}
